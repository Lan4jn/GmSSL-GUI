cmake_minimum_required(VERSION 3.20)
project(gmssl_native_runner C)
set(CMAKE_C_STANDARD 11)
set(GMSSL_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.cache/upstream" CACHE PATH "Pinned GmSSL checkout")
set(GMSSL_EXPECTED_COMMIT "24ae482701a7b124826c382fffc55c19f76d475d")
execute_process(COMMAND git -C "${GMSSL_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE source_commit OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE git_status)
if(NOT git_status EQUAL 0 OR NOT source_commit STREQUAL GMSSL_EXPECTED_COMMIT)
  message(FATAL_ERROR "GmSSL checkout identity mismatch; run native/fetch-source.py")
endif()
execute_process(COMMAND git -C "${GMSSL_SOURCE_DIR}" status --porcelain --untracked-files=no OUTPUT_VARIABLE source_changes OUTPUT_STRIP_TRAILING_WHITESPACE)
if(source_changes)
  message(FATAL_ERROR "GmSSL checkout has tracked modifications")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static runner library" FORCE)
set(ENABLE_SDF OFF CACHE BOOL "Hardware deferred" FORCE)
set(ENABLE_SKF OFF CACHE BOOL "Hardware deferred" FORCE)
set(ENABLE_SM4_CL OFF CACHE BOOL "Portable software profile" FORCE)
set(ENABLE_LMS_CL OFF CACHE BOOL "Portable software profile" FORCE)
set(ENABLE_XMSS_CL OFF CACHE BOOL "Portable software profile" FORCE)

find_package(Python3 REQUIRED COMPONENTS Interpreter)

# Apply explicit, reproducible overlays to build/generated/ preserving pristine upstream checkout
set(GENERATED_SRC_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
execute_process(
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/apply-overlays.py" "${GMSSL_SOURCE_DIR}" "${GENERATED_SRC_DIR}"
  RESULT_VARIABLE overlay_res
)
if(NOT overlay_res EQUAL 0)
  message(FATAL_ERROR "Failed to apply native source overlays")
endif()

add_subdirectory("${GMSSL_SOURCE_DIR}" upstream EXCLUDE_FROM_ALL)

# Overlay replaced sources in gmssl library target
get_target_property(gmssl_sources gmssl SOURCES)
set(updated_gmssl_sources "")
foreach(source IN LISTS gmssl_sources)
  get_filename_component(source_name "${source}" NAME)
  if(EXISTS "${GENERATED_SRC_DIR}/src/${source_name}")
    list(APPEND updated_gmssl_sources "${GENERATED_SRC_DIR}/src/${source_name}")
  else()
    list(APPEND updated_gmssl_sources "${source}")
  endif()
endforeach()
set_target_properties(gmssl PROPERTIES SOURCES "${updated_gmssl_sources}")

get_target_property(tool_sources gmssl-bin SOURCES)
set(runner_sources "${CMAKE_CURRENT_SOURCE_DIR}/runner.c" "${CMAKE_CURRENT_SOURCE_DIR}/extensions.c" "${CMAKE_CURRENT_SOURCE_DIR}/keygen_bridge.c")
foreach(source IN LISTS tool_sources)
  if(NOT source MATCHES "(^|/)gmssl\\.c$")
    get_filename_component(source_name "${source}" NAME)
    if(EXISTS "${GENERATED_SRC_DIR}/tools/${source_name}")
      list(APPEND runner_sources "${GENERATED_SRC_DIR}/tools/${source_name}")
    elseif(IS_ABSOLUTE "${source}")
      list(APPEND runner_sources "${source}")
    else()
      list(APPEND runner_sources "${GMSSL_SOURCE_DIR}/${source}")
    endif()
  endif()
endforeach()
add_executable(gmssl-runner ${runner_sources})
get_directory_property(upstream_definitions DIRECTORY "${GMSSL_SOURCE_DIR}" COMPILE_DEFINITIONS)
target_compile_definitions(gmssl-runner PRIVATE ${upstream_definitions} GMSSL_CLI_SOURCE="${GMSSL_SOURCE_DIR}/tools/gmssl.c" GMSSL_OVERLAYS_APPLIED=1)
target_include_directories(gmssl-runner PRIVATE "${GMSSL_SOURCE_DIR}/include" "${GMSSL_SOURCE_DIR}/tools")
target_link_libraries(gmssl-runner PRIVATE gmssl)
set_target_properties(gmssl-runner PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
install(TARGETS gmssl-runner RUNTIME DESTINATION bin)
